% The two-route pipeline (Figure 4.1), pre-rendered so main.tex does not load TikZ.
\documentclass[12pt,border=2pt]{standalone}
\usepackage[T1]{fontenc}\usepackage{newtxtext}
\usepackage{tikz}\usetikzlibrary{positioning,arrows.meta,calc}
\tikzset{
  box/.style={draw, rounded corners=2pt, align=center, font=\small, minimum height=10mm, text width=27mm, fill=white, line width=0.5pt},
  data/.style={box, fill=blue!10, draw=blue!55},
  proc/.style={box, fill=orange!12, draw=orange!70!black!70},
  res/.style={box, fill=green!12, draw=green!45!black!60},
  arr/.style={-{Latex[length=2mm]}, thick, draw=blue!35!black!55}}
\begin{document}
\begin{tikzpicture}[node distance=5mm and 6mm, box/.append style={minimum height=11mm}]
  \node[data, text width=16mm] (audio) {5\,s audio clip};
  \node[proc, above right=2mm and 8mm of audio] (enc) {MFCC, Whisper\\or XLS-R};
  \node[proc, right=of enc] (aclf) {classifier\\head};
  \node[proc, below right=2mm and 8mm of audio] (asr) {Bangla ASR\\$\rightarrow$ transcript};
  \node[proc, right=of asr] (lclf) {word matching\\or TF-IDF + LR};
  \node[proc, text width=14mm] (fuse) at ($(aclf.east)!0.5!(lclf.east) + (15mm,0)$) {late\\fusion};
  \node[res, right=6mm of fuse, text width=19mm] (out) {dialect\\(9 classes)};
  \draw[arr] (audio.north) |- (enc.west);
  \draw[arr] (enc) -- (aclf);
  \draw[arr] (audio.south) |- (asr.west);
  \draw[arr] (asr) -- (lclf);
  \draw[arr] (aclf.east) -| (fuse.north);
  \draw[arr] (lclf.east) -| (fuse.south);
  \draw[arr] (fuse) -- (out);
  \node[font=\footnotesize\itshape, anchor=south, yshift=0.5mm] at ($(enc.north)!0.5!(aclf.north)$) {acoustic route: the sound};
  \node[font=\footnotesize\itshape, anchor=north, yshift=-0.5mm] at ($(asr.south)!0.5!(lclf.south)$) {lexical route: the words};
\end{tikzpicture}
\end{document}
